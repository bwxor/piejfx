package com.bwxor.piejfx.controller.impl;

import com.bwxor.piejfx.dto.SearchInFilesResult;
import com.bwxor.piejfx.service.SearchInFilesService;
import com.bwxor.piejfx.state.FolderTreeViewState;
import javafx.application.Platform;
import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.concurrent.Task;
import javafx.fxml.FXML;
import javafx.scene.control.*;
import javafx.scene.input.KeyCode;
import javafx.scene.input.KeyEvent;

import java.io.File;
import java.util.List;

public class SearchInFilesViewController {

    @FXML
    private TextField searchTextField;

    @FXML
    private Button searchButton;

    @FXML
    private TableView<SearchInFilesResult> resultsTableView;

    @FXML
    private TableColumn<SearchInFilesResult, String> fileNameColumn;

    @FXML
    private TableColumn<SearchInFilesResult, Integer> lineColumn;

    @FXML
    private TableColumn<SearchInFilesResult, String> textColumn;

    private final SearchInFilesService searchInFilesService;
    private final FolderTreeViewState folderTreeViewState;
    private final ObservableList<SearchInFilesResult> searchResults = FXCollections.observableArrayList();

    private File workspaceRoot;

    public SearchInFilesViewController(SearchInFilesService searchInFilesService, 
                                      FolderTreeViewState folderTreeViewState) {
        this.searchInFilesService = searchInFilesService;
        this.folderTreeViewState = folderTreeViewState;
    }

    @FXML
    public void initialize() {
        // Set up table columns
        fileNameColumn.setCellValueFactory(cellData -> 
            new javafx.beans.property.SimpleStringProperty(cellData.getValue().getFileName()));
        lineColumn.setCellValueFactory(cellData -> 
            new javafx.beans.property.SimpleObjectProperty<>(cellData.getValue().getLineNumber()));
        textColumn.setCellValueFactory(cellData -> 
            new javafx.beans.property.SimpleStringProperty(cellData.getValue().getLineText()));

        // Bind the results list to the table view
        resultsTableView.setItems(searchResults);

        // Set up search button action
        searchButton.setOnAction(event -> performSearch());

        // Set up Enter key handler for search text field
        searchTextField.setOnKeyPressed(this::handleKeyPressed);

        // Auto-set workspace root from the current folder tree state
        if (folderTreeViewState.getOpenedFolder() != null) {
            workspaceRoot = folderTreeViewState.getOpenedFolder();
        }
    }

    private void handleKeyPressed(KeyEvent event) {
        if (event.getCode() == KeyCode.ENTER) {
            performSearch();
        }
    }

    private void performSearch() {
        String searchText = searchTextField.getText();
        
        if (searchText == null || searchText.trim().isEmpty()) {
            showAlert("Search text is empty", "Please enter text to search for.");
            return;
        }

        if (workspaceRoot == null || !workspaceRoot.exists()) {
            showAlert("No workspace selected", "Please open a folder/workspace first.");
            return;
        }

        // Clear previous results
        searchResults.clear();

        // Disable search button during search
        searchButton.setDisable(true);
        searchTextField.setDisable(true);

        // Perform search in background thread
        Task<List<SearchInFilesResult>> searchTask = searchInFilesService.searchInFiles(searchText, workspaceRoot);

        searchTask.setOnSucceeded(event -> {
            final List<SearchInFilesResult> results = searchTask.getValue();
            Platform.runLater(() -> {
                searchResults.addAll(results);
                searchButton.setDisable(false);
                searchTextField.setDisable(false);
                
                if (results.isEmpty()) {
                    showAlert("No results", "No matches found for '" + searchText + "'");
                }
            });
        });

        searchTask.setOnFailed(event -> {
            Platform.runLater(() -> {
                searchButton.setDisable(false);
                searchTextField.setDisable(false);
                showAlert("Search failed", "An error occurred during the search.");
            });
        });

        // Start the task in a background thread
        Thread searchThread = new Thread(searchTask);
        searchThread.setDaemon(true);
        searchThread.start();
    }

    private void showAlert(String title, String message) {
        Alert alert = new Alert(Alert.AlertType.INFORMATION);
        alert.setTitle(title);
        alert.setHeaderText(null);
        alert.setContentText(message);
        alert.showAndWait();
    }

    // Method called by SearchInFilesDialogService
    public void setWorkspaceRoot(File workspaceRoot) {
        this.workspaceRoot = workspaceRoot;
    }

    // Method called by SearchInFilesDialogService to focus the search field
    public void focusSearchField() {
        Platform.runLater(() -> {
            if (searchTextField != null) {
                searchTextField.requestFocus();
            }
        });
    }
}
